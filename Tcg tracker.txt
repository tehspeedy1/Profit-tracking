  const [modal,setModal]=useState(false);const [editId,setEditId]=useState(null);const [del,setDel]=useState(null);const [search,setSearch]=useState("");const [filter,setFilter]=useState("All");
  const blank={productName:"",category:"Booster Box",condition:"Sealed",quantity:"1",costBasis:"",marketValue:"",soldPrice:"",status:"In Stock",platform:"",linkedOrderId:"",notes:"",dateAdded:new Date().toISOString().slice(0,10),dateSold:""};
  const [form,setForm]=useState(blank);const up=(k,v)=>setForm(p=>({...p,[k]:v}));
  const openNew=()=>{setForm(blank);setEditId(null);setModal(true)};const openEdit=i=>{setForm({...blank,...i});setEditId(i.id);setModal(true)};
  const handleSave=()=>{if(!form.productName)return;const entry={...form,id:editId||genId()};const u=editId?inventory.map(i=>i.id===editId?entry:i):[...inventory,entry];setInventory(u);save({inventory:u});setModal(false)};
  const handleDel=()=>{const u=inventory.filter(i=>i.id!==del);setInventory(u);save({inventory:u});setDel(null)};
  const filtered=inventory.filter(i=>{const ms=!search||i.productName.toLowerCase().includes(search.toLowerCase());const mst=filter==="All"||i.status===filter;return ms&&mst});
  const ISC={"In Stock":{bg:"#DBEAFE",text:"#1E40AF",border:"#93C5FD"},Listed:{bg:"#FEF3C7",text:"#92400E",border:"#FCD34D"},Sold:{bg:"#D1FAE5",text:"#065F46",border:"#6EE7B7"},Personal:{bg:"#F3E8FF",text:"#6B21A8",border:"#C4B5FD"}};

  return(<div>
    <div className="toolbar">
      <input className="input" placeholder="Search inventory..." value={search} onChange={e=>setSearch(e.target.value)} style={{flex:"1 1 200px",maxWidth:300}}/>
      <select className="input" value={filter} onChange={e=>setFilter(e.target.value)} style={{flex:"0 0 auto",width:140}}><option value="All">All Status</option><option>In Stock</option><option>Listed</option><option>Sold</option><option>Personal</option></select>
      <button className="btn btn-primary" style={{marginLeft:"auto"}} onClick={openNew}>+ Add Item</button>
    </div>
    {filtered.length===0?<div className="empty"><div className="icon">🗃️</div><p>No inventory items yet.</p></div>:filtered.map(i=>{const profit=i.status==="Sold"?((parseFloat(i.soldPrice)||0)-(parseFloat(i.costBasis)||0)):null;const sc=ISC[i.status]||{};return(
      <div className="card" key={i.id} onClick={()=>openEdit(i)}><div className="card-row"><div><div style={{fontWeight:600,fontSize:15,marginBottom:4}}>{i.productName}</div><div style={{color:"var(--muted)",fontSize:13}}>{i.category} · {i.condition} · Qty: {i.quantity}</div></div><div style={{textAlign:"right"}}><span className="badge" style={{background:sc.bg,color:sc.text,border:"1px solid "+(sc.border||"#444")}}>{i.status}</span><div style={{marginTop:8}}>{i.status==="Sold"?<><div className="mono" style={{color:"#34D399",fontWeight:700,fontSize:14}}>Sold: {fmt$(i.soldPrice)}</div><div style={{color:profit>=0?"#34D399":"#EF4444",fontSize:12,fontWeight:600}}>{profit>=0?"+":""}{fmt$(profit)}</div></>:<div className="mono" style={{color:"var(--accent)",fontWeight:700,fontSize:14}}>Cost: {fmt$(i.costBasis)}</div>}</div></div></div></div>)})}

    <Modal open={modal} onClose={()=>setModal(false)} title={editId?"Edit Item":"New Inventory Item"}>
      <div className="form-grid">
        <div className="full"><Field label="Product Name *"><input className="input" value={form.productName} onChange={e=>up("productName",e.target.value)}/></Field></div>
        <Field label="Category"><select className="input" value={form.category} onChange={e=>up("category",e.target.value)}>{["Booster Box","ETB","Booster Bundle","Collection Box","Tin","Blister Pack","Single Card","Slab/Graded","Accessory","Other"].map(c=><option key={c}>{c}</option>)}</select></Field>
        <Field label="Condition"><select className="input" value={form.condition} onChange={e=>up("condition",e.target.value)}>{COND.map(c=><option key={c}>{c}</option>)}</select></Field>
        <Field label="Quantity"><input type="number" className="input" value={form.quantity} onChange={e=>up("quantity",e.target.value)} min="1"/></Field>
        <Field label="Cost Basis ($)"><input type="number" step="0.01" className="input" value={form.costBasis} onChange={e=>up("costBasis",e.target.value)}/></Field>
        <Field label="Status"><select className="input" value={form.status} onChange={e=>up("status",e.target.value)}><option>In Stock</option><option>Listed</option><option>Sold</option><option>Personal</option></select></Field>
        <Field label="Market Value ($)"><input type="number" step="0.01" className="input" value={form.marketValue} onChange={e=>up("marketValue",e.target.value)}/></Field>
        {form.status==="Sold"&&<><Field label="Sold Price ($)"><input type="number" step="0.01" className="input" value={form.soldPrice} onChange={e=>up("soldPrice",e.target.value)}/></Field><Field label="Date Sold"><input type="date" className="input" value={form.dateSold} onChange={e=>up("dateSold",e.target.value)}/></Field></>}
        <Field label="Platform"><select className="input" value={form.platform} onChange={e=>up("platform",e.target.value)}><option value="">— None —</option>{["eBay","TCGPlayer","Mercari","Facebook Marketplace","Local Sale","Other"].map(p=><option key={p}>{p}</option>)}</select></Field>
        <Field label="Linked Order"><select className="input" value={form.linkedOrderId} onChange={e=>up("linkedOrderId",e.target.value)}><option value="">— None —</option>{orders.map(o=><option key={o.id} value={o.id}>#{o.orderNumber}</option>)}</select></Field>

      </div>

        {editId?<button className="btn-danger" onClick={()=>{setModal(false);setDel(editId)}}>Delete</button>:<div/>}

      </div>
    </Modal>
    <Confirm open={!!del} onClose={()=>setDel(null)} onConfirm={handleDel} message="Delete this item?"/>
  </div>);
}

// ─── Profits ───
function ProfitsTab({orders}){
  const sold=orders.filter(o=>o.soldPrice&&parseFloat(o.soldPrice)>0);
  const rev=sold.reduce((s,o)=>s+(parseFloat(o.soldPrice)||0),0);
  const cogs=sold.reduce((s,o)=>s+(parseFloat(o.totalCost)||0),0);
  const profit=rev-cogs;
  const margin=rev>0?((profit/rev)*100).toFixed(1):"0.0";
  const totalInv=orders.reduce((s,o)=>s+(parseFloat(o.totalCost)||0),0);
  const roi=totalInv>0?((profit/totalInv)*100).toFixed(1):"0.0";
  const unsoldCost=orders.filter(o=>!o.soldPrice||parseFloat(o.soldPrice)===0).reduce((s,o)=>s+(parseFloat(o.totalCost)||0),0);

  const byPlat={};sold.forEach(o=>{const p=o.soldPlatform||"Unknown";if(!byPlat[p])byPlat[p]={rev:0,profit:0,count:0};byPlat[p].rev+=parseFloat(o.soldPrice)||0;byPlat[p].profit+=(parseFloat(o.soldPrice)||0)-(parseFloat(o.totalCost)||0);byPlat[p].count+=1});
  const bySrc={};sold.forEach(o=>{const s=o.source||"Unknown";if(!bySrc[s])bySrc[s]={rev:0,profit:0,count:0};bySrc[s].rev+=parseFloat(o.soldPrice)||0;bySrc[s].profit+=(parseFloat(o.soldPrice)||0)-(parseFloat(o.totalCost)||0);bySrc[s].count+=1});

  return(<div>
    <div className="stats">
      <StatCard label="TOTAL REVENUE" value={fmt$(rev)} icon="💵" color="#34D399"/>
      <StatCard label="TOTAL COGS" value={fmt$(cogs)} icon="💳" color="#F472B6"/>
      <StatCard label="NET PROFIT" value={fmt$(profit)} icon="📈" color={profit>=0?"#34D399":"#EF4444"}/>
      <StatCard label="MARGIN" value={margin+"%"} icon="📊" color="#818CF8"/>
    </div>
    <div className="stats">
      <StatCard label="ITEMS SOLD" value={sold.length} icon="✅" color="#FBBF24"/>
      <StatCard label="UNSOLD COST" value={fmt$(unsoldCost)} icon="🗃️" color="#FB923C"/>
      <StatCard label="TOTAL INVESTED" value={fmt$(totalInv)} icon="💰" color="#818CF8"/>
      <StatCard label="ROI" value={roi+"%"} icon="🎯" color={parseFloat(roi)>=0?"#34D399":"#EF4444"}/>
    </div>
    {Object.keys(byPlat).length>0&&<div className="section-card"><h3>Profit by Selling Platform</h3>{Object.entries(byPlat).sort((a,b)=>b[1].profit-a[1].profit).map(([p,d])=><div className="section-row" key={p}><div><div style={{color:"#C8C8E8",fontSize:14,fontWeight:500}}>{p}</div><div style={{color:"var(--dim)",fontSize:12}}>{d.count} sales · {fmt$(d.rev)} rev</div></div><div className="mono" style={{color:d.profit>=0?"#34D399":"#EF4444",fontWeight:700,fontSize:15}}>{d.profit>=0?"+":""}{fmt$(d.profit)}</div></div>)}</div>}
    {Object.keys(bySrc).length>0&&<div className="section-card"><h3>Profit by Source</h3>{Object.entries(bySrc).sort((a,b)=>b[1].profit-a[1].profit).map(([s,d])=><div className="section-row" key={s}><div><div style={{color:"#C8C8E8",fontSize:14,fontWeight:500}}>{s}</div><div style={{color:"var(--dim)",fontSize:12}}>{d.count} sales · {fmt$(d.rev)} rev</div></div><div className="mono" style={{color:d.profit>=0?"#34D399":"#EF4444",fontWeight:700,fontSize:15}}>{d.profit>=0?"+":""}{fmt$(d.profit)}</div></div>)}</div>}
    {sold.length>0?<div className="section-card"><h3>Sales History</h3>{[...sold].sort((a,b)=>new Date(b.dateSold||b.orderDate)-new Date(a.dateSold||a.orderDate)).map(o=>{const p=(parseFloat(o.soldPrice)||0)-(parseFloat(o.totalCost)||0);return <div className="section-row" key={o.id}><div><div style={{color:"#C8C8E8",fontSize:14,fontWeight:500}}>{o.productName}</div><div style={{color:"var(--dim)",fontSize:12}}>#{o.orderNumber} · {o.soldPlatform||"—"} · Sold {fmtDate(o.dateSold)} · Cost: {fmt$(o.totalCost)}</div></div><div style={{textAlign:"right"}}><div className="mono" style={{color:"#C8C8E8",fontWeight:600,fontSize:14}}>{fmt$(o.soldPrice)}</div><div style={{color:p>=0?"#34D399":"#EF4444",fontSize:12,fontWeight:600}}>{p>=0?"+":""}{fmt$(p)}</div></div></div>})}</div>:<div className="empty"><div className="icon">💰</div><p>No sales yet. Add a "Sold For" price to any order.</p></div>}
  </div>);
}

// ─── Main App ───
function App(){
  const [tab,setTab]=useState("dashboard");
  const [orders,setOrders]=useState([]);
  const [inventory,setInventory]=useState([]);
  const [loaded,setLoaded]=useState(false);
  const [msg,setMsg]=useState("");

  useEffect(()=>{
    const data=loadData();
    let o=data.orders||[];let inv=data.inventory||[];
    const existing=new Set(o.map(x=>x.orderNumber));
    const newO=GMAIL_ORDERS.filter(x=>!existing.has(x.orderNumber));
    if(newO.length>0){o=[...o,...newO];saveDataLS({orders:o,inventory:inv});}
    setOrders(o);setInventory(inv);setLoaded(true);
  },[]);

  const save=useCallback((overrides={})=>{
    const data={orders:overrides.orders||orders,inventory:overrides.inventory||inventory};
    saveDataLS(data);
  },[orders,inventory]);

  const exportData=()=>{const d=JSON.stringify({orders,inventory,exportedAt:new Date().toISOString()},null,2);navigator.clipboard.writeText(d).then(()=>{setMsg("Copied to clipboard!");setTimeout(()=>setMsg(""),3000)}).catch(()=>{const w=window.open();w.document.write("<pre>"+d+"</pre>")})};

const resetAll=()=>{if(confirm(“Delete ALL data permanently?”)){setOrders([]);setInventory([]);saveDataLS({orders:[],inventory:[]});setMsg(“Cleared.”);setTimeout(()=>setMsg(””),3000)}};

if(!loaded)return <div style={{minHeight:“100vh”,display:“flex”,alignItems:“center”,justifyContent:“center”,background:“var(–bg)”}}><div style={{color:“var(–accent)”,fontSize:18}}>Loading…</div></div>;

return(<div>
<div className="header"><div className="header-inner"><span style={{fontSize:28}}>⚡</span><div><h1>TCG Tracker</h1><div className="sub">POKEMON RESALE MANAGER</div></div></div></div>
<div className="tabs"><div className="tabs-inner">{TABS.map(t=><button key={t} className={“tab-btn”+(tab===t?” active”:””)} onClick={()=>setTab(t)}><span className="ico">{TAB_ICONS[t]}</span>{TAB_LABELS[t]}</button>)}</div></div>
<div className="content">
{tab===“dashboard”&&<Dashboard orders={orders}/>}
{tab===“orders”&&<OrdersTab orders={orders} setOrders={setOrders} save={save}/>}
{tab===“inventory”&&<InventoryTab inventory={inventory} setInventory={setInventory} orders={orders} save={save}/>}
{tab===“profits”&&<ProfitsTab orders={orders}/>}
<div className="data-footer">
<button className=“btn btn-secondary” style={{fontSize:12,padding:“8px 16px”}} onClick={exportData}>Export</button>
<button className="btn-danger" style={{fontSize:12}} onClick={resetAll}>Reset All</button>
{msg&&<span style={{color:”#34D399”,fontSize:13,marginLeft:8}}>{msg}</span>}
</div>
</div>

  </div>);
}

ReactDOM.render(<App/>,document.getElementById(“app”));
</script>
